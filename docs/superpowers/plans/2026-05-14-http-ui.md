# Coax Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a cross-platform Electron desktop app that imports, edits, and executes `.http` files with workspace-level collections, layered environments, encrypted secrets, and shareable export.

**Architecture:** Strict main/renderer Electron split. Renderer is pure Melodic web components UI; main owns SQLite (via `better-sqlite3`), the `.http` parser, the variable resolver, secret encryption (`safeStorage`), and a Node Worker that executes HTTP requests with `undici`. Renderer talks to main via a single typed IPC channel.

**Tech Stack:** Electron, TypeScript (strict), `@melodicdev/core` 1.6.x, `@melodicdev/components` 1.6.x, Monaco editor, `undici`, `better-sqlite3`, `jsonpath-plus`, Vitest, Playwright, electron-vite, electron-builder.

**Reference spec:** `docs/superpowers/specs/2026-05-14-http-ui-design.md`

```ts
import type { Line, HttpMethod } from './types.js';

const METHODS = new Set<HttpMethod>(['GET', 'POST', 'PUT', 'PATCH', 'DELETE', 'HEAD', 'OPTIONS']);

const RX = {
  separator: /^###\s*(.*)$/,
  variable: /^@([A-Za-z_][\w-]*)\s*=\s*(.*)$/,
  name: /^#\s*@name\s+(\S+)\s*$/,
  graphql: /^#\s*@graphql\s*$/,
  fileBody: /^<\s+(\S.*)$/,
  request: /^([A-Z]+)\s+(\S+)(?:\s+(HTTP\/[\d.]+))?\s*$/,
  header: /^([A-Za-z0-9][A-Za-z0-9_-]*)\s*:\s*(.*)$/,
  comment: /^(?:#|\/\/)(.*)$/,
};

export function lex(input: string): Line[] {
  return input.split(/\r?\n/).map((raw, i): Line => {
    const lineNo = i + 1;
    if (raw.trim() === '') return { kind: 'blank', lineNo };

    let m;
    if ((m = RX.separator.exec(raw))) return { kind: 'separator', title: m[1].trim(), lineNo };
    if ((m = RX.name.exec(raw))) return { kind: 'name', name: m[1], lineNo };
    if ((m = RX.graphql.exec(raw))) return { kind: 'graphql', lineNo };
    if ((m = RX.variable.exec(raw))) return { kind: 'variable', name: m[1], value: m[2].trim(), lineNo };
    if ((m = RX.fileBody.exec(raw))) return { kind: 'fileBody', path: m[1].trim(), lineNo };
    if ((m = RX.request.exec(raw))) {
      const method = m[1] as HttpMethod;
      if (METHODS.has(method)) return { kind: 'request', method, url: m[2], httpVersion: m[3], lineNo };
    }
    if ((m = RX.header.exec(raw))) return { kind: 'header', key: m[1], value: m[2], lineNo };
    if ((m = RX.comment.exec(raw))) return { kind: 'comment', text: m[1], lineNo };
    return { kind: 'text', text: raw, lineNo };
  });
}
```

```ts
import { lex } from './lexer.js';
import type { ParsedFile, ParsedRequest, VarDef, Header, BodyKind, Line } from './types.js';

function inferBodyKind(headers: Header[], graphqlHint: boolean): BodyKind {
  if (graphqlHint) return 'graphql';
  const ct = headers.find(h => h.key.toLowerCase() === 'content-type')?.value.toLowerCase() ?? '';
  if (ct.startsWith('multipart/')) return 'multipart';
  if (ct.includes('json')) return 'json';
  if (ct.includes('x-www-form-urlencoded')) return 'form';
  if (ct.startsWith('text/') || ct.includes('xml')) return 'text';
  return 'text';
}

export function parseHttpFile(input: string): ParsedFile {
  const lines = lex(input);
  const variables: VarDef[] = [];
  const requests: ParsedRequest[] = [];

  // Top-level variables: any `@var =` lines that appear before the first separator/request line
  let firstReqIdx = lines.findIndex(l => l.kind === 'separator' || l.kind === 'request');
  if (firstReqIdx === -1) firstReqIdx = lines.length;
  for (const l of lines.slice(0, firstReqIdx)) {
    if (l.kind === 'variable') variables.push({ name: l.name, value: l.value, line: l.lineNo });
  }

  // Group remaining lines into request blocks bounded by separators / start of file
  type Block = { startLine: number; endLine: number; title: string; lines: Line[] };
  const blocks: Block[] = [];
  let current: Block | null = null;
  for (let i = firstReqIdx; i < lines.length; i++) {
    const l = lines[i]!;
    if (l.kind === 'separator') {
      if (current) blocks.push(current);
      current = { startLine: l.lineNo, endLine: l.lineNo, title: l.title, lines: [] };
      continue;
    }
    if (!current) current = { startLine: l.lineNo, endLine: l.lineNo, title: '', lines: [] };
    current.lines.push(l);
    current.endLine = l.lineNo;
  }
  if (current) blocks.push(current);

  for (const b of blocks) {
    const req = blockToRequest(b);
    if (req) requests.push(req);
  }
  return { variables, requests };
}

function blockToRequest(b: { startLine: number; endLine: number; title: string; lines: Line[] }): ParsedRequest | null {
  let name: string | undefined;
  let graphqlHint = false;
  let fileHint: string | undefined;
  let methodIdx = -1;
  for (let i = 0; i < b.lines.length; i++) {
    const l = b.lines[i]!;
    if (l.kind === 'name') name = l.name;
    else if (l.kind === 'graphql') graphqlHint = true;
    else if (l.kind === 'request') { methodIdx = i; break; }
  }
  if (methodIdx === -1) return null;
  const reqLine = b.lines[methodIdx] as Extract<Line, { kind: 'request' }>;

  const headers: Header[] = [];
  let bodyStart = -1;
  for (let i = methodIdx + 1; i < b.lines.length; i++) {
    const l = b.lines[i]!;
    if (l.kind === 'header') headers.push({ key: l.key, value: l.value });
    else if (l.kind === 'blank') { bodyStart = i + 1; break; }
    else if (l.kind === 'request' || l.kind === 'separator') break;
    // ignore comments inside header block
  }

  const bodyParts: string[] = [];
  if (bodyStart >= 0) {
    for (let i = bodyStart; i < b.lines.length; i++) {
      const l = b.lines[i]!;
      if (l.kind === 'fileBody') { fileHint = l.path; continue; }
      if (l.kind === 'blank') { bodyParts.push(''); continue; }
      if (l.kind === 'text') bodyParts.push(l.text);
      else if (l.kind === 'header') bodyParts.push(`${l.key}: ${l.value}`);
      else if (l.kind === 'comment') continue;
      else if (l.kind === 'request') bodyParts.push(`${l.method} ${l.url}`);
      else if (l.kind === 'variable') bodyParts.push(`@${l.name} = ${l.value}`);
    }
  }
  const raw = bodyParts.join('\n').replace(/\n+$/, '');
  const kind = raw === '' && !fileHint ? 'none' : inferBodyKind(headers, graphqlHint);

  return {
    ...(name !== undefined ? { name } : {}),
    title: b.title,
    method: reqLine.method,
    url: reqLine.url,
    ...(reqLine.httpVersion !== undefined ? { httpVersion: reqLine.httpVersion } : {}),
    headers,
    ...(raw !== '' || fileHint
      ? { body: { kind, raw, ...(kind === 'multipart' ? { parts: [] } : {}) } }
      : {}),
    hints: {
      ...(graphqlHint ? { graphql: true as const } : {}),
      ...(fileHint !== undefined ? { file: fileHint } : {}),
      ...(headers.find(h => h.key.toLowerCase() === 'content-type')?.value
        ? { contentType: headers.find(h => h.key.toLowerCase() === 'content-type')!.value }
        : {}),
    },
    range: { startLine: b.startLine, endLine: b.endLine },
  };
}
```

```ts
import type { MultipartPart, Header } from './types.js';

export function boundaryFromContentType(ct: string | undefined): string | undefined {
  if (!ct) return undefined;
  const m = /boundary=("?)([^";]+)\1/i.exec(ct);
  return m?.[2];
}

export function splitMultipart(body: string, boundary: string): MultipartPart[] {
  const sep = `--${boundary}`;
  const sections = body.split(sep).slice(1);
  const parts: MultipartPart[] = [];
  for (const raw of sections) {
    if (raw.startsWith('--')) break;
    const cleaned = raw.replace(/^\r?\n/, '').replace(/\r?\n$/, '');
    const idx = cleaned.search(/\r?\n\r?\n/);
    if (idx === -1) continue;
    const headerBlock = cleaned.slice(0, idx);
    const bodyBlock = cleaned.slice(idx).replace(/^\r?\n\r?\n/, '');
    const headers: Header[] = [];
    let filename: string | undefined;
    for (const hl of headerBlock.split(/\r?\n/)) {
      const m = /^([A-Za-z0-9-]+)\s*:\s*(.*)$/.exec(hl);
      if (!m) continue;
      headers.push({ key: m[1], value: m[2] });
      if (m[1].toLowerCase() === 'content-disposition') {
        const fn = /filename="?([^";]+)"?/i.exec(m[2]);
        if (fn) filename = fn[1];
      }
    }
    parts.push({ headers, body: bodyBlock, ...(filename !== undefined ? { filename } : {}) });
  }
  return parts;
}
```

```ts
import type { ParsedFile, ParsedRequest, VarDef } from './types.js';

/**
 * Two modes:
 *  - serializeHttpFile(file, originalText): preserves original byte-for-byte for
 *    untouched ranges; only re-emits requests whose range was edited.
 *  - serializeHttpFile(file): canonical emit (used for brand-new collections).
 */
export function serializeHttpFile(file: ParsedFile, original?: string): string {
  if (original === undefined) return canonical(file);
  // Preservation mode: walk original lines, replace each request's range with
  // a fresh emit only when its content differs.
  const lines = original.split(/\r?\n/);
  const out: string[] = [];
  let i = 0;
  const reqByStart = new Map(file.requests.map(r => [r.range.startLine, r]));
  while (i < lines.length) {
    const lineNo = i + 1;
    const req = reqByStart.get(lineNo);
    if (req) {
      const original = lines.slice(req.range.startLine - 1, req.range.endLine).join('\n');
      const fresh = emitRequest(req);
      out.push(originalEqualsFresh(original, fresh) ? original : fresh);
      i = req.range.endLine;
      continue;
    }
    out.push(lines[i] ?? '');
    i++;
  }
  return out.join('\n');
}

function originalEqualsFresh(a: string, b: string): boolean {
  // Identity comparison for unchanged regions; rendered identically in both modes.
  return a === b;
}

function canonical(file: ParsedFile): string {
  const out: string[] = [];
  for (const v of file.variables) out.push(`@${v.name} = ${v.value}`);
  if (file.variables.length) out.push('');
  for (const r of file.requests) {
    out.push(emitRequest(r));
    out.push('');
  }
  return out.join('\n');
}

function emitRequest(r: ParsedRequest): string {
  const parts: string[] = [];
  parts.push(`### ${r.title}`.trimEnd());
  if (r.name) parts.push(`# @name ${r.name}`);
  if (r.hints.graphql) parts.push('# @graphql');
  parts.push(`${r.method} ${r.url}${r.httpVersion ? ` ${r.httpVersion}` : ''}`);
  for (const h of r.headers) parts.push(`${h.key}: ${h.value}`);
  if (r.body && (r.body.raw !== '' || r.hints.file)) {
    parts.push('');
    if (r.hints.file) parts.push(`< ${r.hints.file}`);
    else parts.push(r.body.raw);
  }
  return parts.join('\n');
}
```

```ts
import type { ResolverContext, ResolveResult } from './types.js';

const TOKEN = /\{\{\s*([^}]+?)\s*\}\}/g;

export function resolve(input: string, ctx: ResolverContext): ResolveResult {
  const unresolved = new Set<string>();
  const text = input.replace(TOKEN, (whole, expr: string) => {
    const value = lookup(expr.trim(), ctx);
    if (value === undefined) {
      unresolved.add(expr.trim());
      return whole;
    }
    return value;
  });
  return { text, unresolved: [...unresolved] };
}

function lookup(expr: string, ctx: ResolverContext): string | undefined {
  if (expr.startsWith('$')) return builtin(expr, ctx);
  // response chain handled in Task 2.2
  if (expr.includes('.response.')) return undefined;
  const s = ctx.scopes;
  return s.request?.[expr] ?? s.collectionEnv?.[expr] ?? s.globalEnv?.[expr] ?? s.collectionDefaults?.[expr];
}

function builtin(expr: string, ctx: ResolverContext): string | undefined {
  const now = ctx.now?.() ?? new Date();
  const rand = ctx.random ?? Math.random;
  if (expr === '$timestamp') return String(Math.floor(now.getTime() / 1000));
  if (expr === '$isoTimestamp') return now.toISOString();
  if (expr === '$guid') return guid(rand);
  const m = /^\$randomInt\s+(-?\d+)\s+(-?\d+)$/.exec(expr);
  if (m) {
    const lo = Number(m[1]);
    const hi = Number(m[2]);
    return String(Math.floor(rand() * (hi - lo)) + lo);
  }
  return undefined;
}

function guid(rand: () => number): string {
  const hex = (n: number) => Math.floor(rand() * 16 ** n).toString(16).padStart(n, '0');
  return `${hex(8)}-${hex(4)}-4${hex(3)}-${(8 + Math.floor(rand() * 4)).toString(16)}${hex(3)}-${hex(12)}`;
}
```

```ts
import { JSONPath } from 'jsonpath-plus';

function chain(expr: string, ctx: ResolverContext): string | undefined {
  const m = /^([A-Za-z_]\w*)\.response\.(body|headers)\.(.+)$/.exec(expr);
  if (!m) return undefined;
  const [, name, kind, rest] = m;
  const r = ctx.responses?.[name];
  if (!r) return undefined;
  if (kind === 'headers') return r.headers[rest.toLowerCase()];
  const path = rest.startsWith('$') ? rest : `$.${rest}`;
  const result = JSONPath({ path, json: r.body, wrap: false }) as unknown;
  if (result === undefined || result === null) return undefined;
  return typeof result === 'string' ? result : JSON.stringify(result);
}
```

```ts
import Database from 'better-sqlite3';
import { readFileSync, readdirSync } from 'node:fs';
import { join, dirname } from 'node:path';
import { fileURLToPath } from 'node:url';

const __dirname = dirname(fileURLToPath(import.meta.url));

export type Db = Database.Database;

export function openDb(path: string): Db {
  const db = new Database(path);
  db.pragma('journal_mode = WAL');
  db.pragma('foreign_keys = ON');
  migrate(db);
  return db;
}

function migrate(db: Db): void {
  const has = db.prepare("SELECT name FROM sqlite_master WHERE type='table' AND name='schema_version'").get();
  const current = has ? (db.prepare('SELECT version FROM schema_version').get() as { version: number }).version : 0;
  const dir = join(__dirname, 'migrations');
  const files = readdirSync(dir).filter(f => f.endsWith('.sql')).sort();
  for (const f of files) {
    const v = Number(f.split('_')[0]);
    if (v <= current) continue;
    db.exec(readFileSync(join(dir, f), 'utf8'));
  }
}
```

```ts
import { parentPort } from 'node:worker_threads';
import { request } from 'undici';
import type { RequestSpec, RunnerResult, WorkerInbound, WorkerOutbound } from './types.js';

const controllers = new Map<string, AbortController>();

parentPort!.on('message', async (msg: WorkerInbound) => {
  if (msg.kind === 'cancel') { controllers.get(msg.id)?.abort(); return; }
  if (msg.kind !== 'send') return;
  const result = await runOne(msg.spec);
  const out: WorkerOutbound = { kind: 'result', result };
  parentPort!.postMessage(out);
});

async function runOne(spec: RequestSpec): Promise<RunnerResult> {
  const ac = new AbortController();
  controllers.set(spec.id, ac);
  const t0 = performance.now();
  const timer = spec.timeoutMs ? setTimeout(() => ac.abort(new Error('TIMEOUT')), spec.timeoutMs) : null;
  try {
    const res = await request(spec.url, {
      method: spec.method,
      headers: spec.headers,
      body: spec.body?.raw,
      signal: ac.signal,
    });
    const buf = Buffer.from(await res.body.arrayBuffer());
    const headers: Record<string, string> = {};
    for (const [k, v] of Object.entries(res.headers)) headers[k] = Array.isArray(v) ? v.join(', ') : (v ?? '');
    return { id: spec.id, ok: true, status: res.statusCode, headers, bodyBytes: new Uint8Array(buf), ms: Math.round(performance.now() - t0), sizeBytes: buf.byteLength };
  } catch (e: unknown) {
    return { id: spec.id, ok: false, category: classify(e), message: (e as Error).message };
  } finally {
    if (timer) clearTimeout(timer);
    controllers.delete(spec.id);
  }
}

function classify(e: unknown): 'network'|'tls'|'timeout'|'aborted'|'invalid'|'unknown' {
  const m = (e as Error)?.message ?? '';
  if (m.includes('TIMEOUT')) return 'timeout';
  if (m.includes('aborted') || m.includes('AbortError')) return 'aborted';
  if (m.includes('CERT') || m.includes('TLS')) return 'tls';
  if (m.includes('ENOTFOUND') || m.includes('ECONN') || m.includes('EAI_AGAIN')) return 'network';
  if (m.includes('Invalid URL')) return 'invalid';
  return 'unknown';
}
```

```ts
import { Worker } from 'node:worker_threads';

let worker: Worker | null = null;
const pending = new Map<string, (r: RunnerResult) => void>();

export async function startRunner(): Promise<void> {
  if (worker) return;
  worker = new Worker(join(__dirname, 'worker.js'));
  worker.on('message', (m: WorkerOutbound) => {
    if (m.kind === 'result') pending.get(m.result.id)?.(m.result);
    pending.delete(m.result.id);
  });
}
export async function stopRunner(): Promise<void> {
  await worker?.terminate(); worker = null; pending.clear();
}
export function send(spec: RequestSpec): Promise<RunnerResult> {

    pending.set(spec.id, resolve);
    worker!.postMessage({ kind: 'send', spec });
  });
}
```

```ts
import type { IpcRequest, IpcResponse } from './types.js';

export type Handlers = { [K in IpcRequest['kind']]?: (msg: Extract<IpcRequest, { kind: K }>) => unknown | Promise<unknown> };

export function createDispatcher(handlers: Handlers) {
  return async (msg: IpcRequest): Promise<IpcResponse> => {
    const h = handlers[msg.kind] as ((m: IpcRequest) => unknown | Promise<unknown>) | undefined;
    if (!h) return { ok: false, error: { code: 'UNKNOWN_KIND', message: `No handler for ${msg.kind}` } };
    try {
      const data = await h(msg);
      return { ok: true, data };
    } catch (e: unknown) {
      const err = e as Error;
      return { ok: false, error: { code: err.message, message: err.message } };
    }
  };
}
```

```ts
class MonacoEl extends HTMLElement {
  static observedAttributes = ['value', 'language', 'readonly'];

      readOnly: this.hasAttribute('readonly'),
      automaticLayout: true,
      minimap: { enabled: false },
    });

    if (n === 'value' && this.editor && this.editor.getValue() !== v) this.editor.setValue(v ?? '');

  }
}
```

```ts
export function toCurl(draft: { method: string; url: string; headers: { key: string; value: string }[]; body?: { raw: string } }, secretMaskedHeaders: Set<string> = new Set()): string {
  const parts = [`curl -X ${draft.method} ${shellQuote(draft.url)}`];
  for (const h of draft.headers) parts.push(`  -H ${shellQuote(`${h.key}: ${secretMaskedHeaders.has(h.key.toLowerCase()) ? '••••' : h.value}`)}`);
  if (draft.body?.raw) parts.push(`  --data ${shellQuote(draft.body.raw)}`);
  return parts.join(' \\\n');
}
function shellQuote(s: string): string { return `'${s.replace(/'/g, `'\\''`)}'`; }
```
